    private void createTablePanel() {
        tablePanel.setBackground(Color.WHITE);
        
        // Sửa lại columns để hiển thị đầy đủ thông tin sản phẩm
        String[] columnNames = {"ID", "Tên sản phẩm", "Danh mục", "Đơn giá", "Số lượng", 
                               "Thời gian CB (phút)", "Mang về", "Tại quán", "Trạng thái"};
        
        tableModel = new DefaultTableModel(columnNames, 0) {
            @Override 
            public boolean isCellEditable(int row, int column) { 
                return false; 
            }
            
            @Override
            public Class<?> getColumnClass(int columnIndex) {
                switch (columnIndex) {
                    case 0: return Integer.class;    // ID
                    case 3: return String.class;     // Đơn giá (formatted)
                    case 4: return Integer.class;    // Số lượng
                    case 5: return Integer.class;    // Thời gian
                    case 6: case 7: return String.class; // Boolean as String
                    default: return String.class;
                }
            }
        };
        
        table = new JTable(tableModel);
        table.setAutoResizeMode(JTable.AUTO_RESIZE_OFF);
        table.setSelectionMode(ListSelectionModel.SINGLE_SELECTION);
        table.setRowHeight(32);
        table.setGridColor(new Color(200, 200, 200));
        table.setFont(new Font("Arial", Font.PLAIN, 12));
        table.setBackground(Color.WHITE);
        table.setSelectionBackground(secondaryColor);
        table.setSelectionForeground(Color.BLACK);
        table.setShowVerticalLines(true);
        table.setShowHorizontalLines(true);
        
        // Header styling
        table.getTableHeader().setFont(new Font("Arial", Font.BOLD, 12));
        table.getTableHeader().setBackground(primaryColor);
        table.getTableHeader().setForeground(Color.WHITE);
        table.getTableHeader().setPreferredSize(new Dimension(0, 35));
        table.getTableHeader().setBorder(BorderFactory.createRaisedBorder());
        
        // Set column widths
        int[] columnWidths = {50, 180, 120, 100, 80, 100, 80, 80, 100};
        for (int i = 0; i < columnWidths.length; i++) {
            table.getColumnModel().getColumn(i).setPreferredWidth(columnWidths[i]);
        }
        
        // Center alignment for numeric columns
        DefaultTableCellRenderer centerRenderer = new DefaultTableCellRenderer();
        centerRenderer.setHorizontalAlignment(JLabel.CENTER);
        table.getColumnModel().getColumn(0).setCellRenderer(centerRenderer); // ID
        table.getColumnModel().getColumn(3).setCellRenderer(centerRenderer); // Đơn giá
        table.getColumnModel().getColumn(4).setCellRenderer(centerRenderer); // Số lượng
        table.getColumnModel().getColumn(5).setCellRenderer(centerRenderer); // Thời gian
        table.getColumnModel().getColumn(6).setCellRenderer(centerRenderer); // Mang về
        table.getColumnModel().getColumn(7).setCellRenderer(centerRenderer); // Tại quán
        
        // Custom renderer for status column
        DefaultTableCellRenderer statusRenderer = new DefaultTableCellRenderer() {
            @Override
            public Component getTableCellRendererComponent(JTable table, Object value,
                    boolean isSelected, boolean hasFocus, int row, int column) {
                Component c = super.getTableCellRendererComponent(table, value, isSelected, hasFocus, row, column);
                if (!isSelected) {
                    if ("Còn".equals(value)) {
                        c.setBackground(new Color(220, 255, 220));
                        c.setForeground(new Color(0, 120, 0));
                    } else {
                        c.setBackground(new Color(255, 220, 220));
                        c.setForeground(new Color(120, 0, 0));
                    }
                }
                setHorizontalAlignment(JLabel.CENTER);
                return c;
            }
        };
        table.getColumnModel().getColumn(8).setCellRenderer(statusRenderer);
        
        table.addMouseListener(new MouseAdapter() {
            @Override 
            public void mouseClicked(MouseEvent e) {
                int selectedRow = table.getSelectedRow();
                if (selectedRow >= 0) displaySelected(selectedRow);
            }
        });
        
        JScrollPane scrollPane = new JScrollPane(table);
        scrollPane.setBorder(BorderFactory.createLineBorder(primaryColor, 1));
        scrollPane.getViewport().setBackground(Color.WHITE);
        tablePanel.add(scrollPane, BorderLayout.CENTER);

    private JButton createStyledButton(String text, Color bgColor) {
        JButton button = new JButton(text);
        button.setFont(new Font("Arial", Font.BOLD, 13));
        button.setBackground(bgColor);
        button.setForeground(Color.WHITE);
        button.setFocusPainted(false);
        button.setBorder(BorderFactory.createEmptyBorder(10, 20, 10, 20));
        button.setCursor(new Cursor(Cursor.HAND_CURSOR));
        
        // Hover effect
        button.addMouseListener(new MouseAdapter() {
            @Override
            public void mouseEntered(MouseEvent e) {
                button.setBackground(bgColor.darker());
            }
            
            @Override
            public void mouseExited(MouseEvent e) {
                button.setBackground(bgColor);
            }
        });
        
        return button;
    }

    private void loadData() {
        new SwingWorker<Void, Void>() {
            @Override
            protected Void doInBackground() throws Exception {
                progressBar.setVisible(true);
                List<SanPham> list = controller.getAllSanPham();
                publish();
                return null;
            }

            @Override
            protected void done() {
                progressBar.setVisible(false);
                List<SanPham> list = controller.getAllSanPham();
                displayList(list);
            }
        }.execute();
    }

    private void loadCategories() {
        loadCategories(cboDanhMuc);
        loadCategories(cboDanhMucSearch);
    }

    private void loadCategories(JComboBox<String> comboBox) {
        try (Connection conn = DatabaseUtil.getConnection();
             PreparedStatement ps = conn.prepareStatement("SELECT id_danh_muc, ten_danh_muc FROM DanhMucSanPham WHERE trang_thai = 'Hoạt động'")) {
            ResultSet rs = ps.executeQuery();
            comboBox.removeAllItems();
            if (comboBox == cboDanhMucSearch) comboBox.addItem("Tất cả");
            else comboBox.addItem("Chọn danh mục");
            while (rs.next()) {
                comboBox.addItem(rs.getInt("id_danh_muc") + " - " + rs.getString("ten_danh_muc"));
            }
        } catch (SQLException e) {
            e.printStackTrace();
            JOptionPane.showMessageDialog(this, "Lỗi tải danh mục: " + e.getMessage());
        }
    }

    // Sửa lại phương thức displayList để hiển thị đầy đủ thông tin
    private void displayList(List<SanPham> list) {
        tableModel.setRowCount(0);
        for (SanPham sp : list) {
            Object[] row = {
                sp.getIdSanPham(),                                          // ID
                sp.getTenSanPham(),                                         // Tên sản phẩm
                sp.getTenDanhMuc() != null ? sp.getTenDanhMuc() : "N/A",    // Danh mục
                formatCurrency(sp.getDonGia()),                             // Đơn giá
                sp.getSoLuong(),                                            // Số lượng
                sp.getThoiGianChuanBi(),                                     // Thời gian chuẩn bị
                sp.isCoBanMangVe() ? "Có" : "Không",                        // Mang về
                sp.isCoBanTaiQuan() ? "Có" : "Không",                       // Tại quán
                sp.getTrangThai()                                           // Trạng thái
            };
            tableModel.addRow(row);
        }
    }
    
    private String formatCurrency(BigDecimal amount) {
        if (amount == null) return "0 VNĐ";
        return String.format("%,.0f VNĐ", amount.doubleValue());
    }

    // Sửa lại phương thức displaySelected để lấy đúng dữ liệu từ bảng
    private void displaySelected(int row) {
        // Lấy ID từ cột đầu tiên
        Object idObj = tableModel.getValueAt(row, 0);
        if (idObj != null) {
            txtId.setText(idObj.toString());
            
            // Tìm sản phẩm theo ID để lấy đầy đủ thông tin
            int id = Integer.parseInt(idObj.toString());
            List<SanPham> allProducts = controller.getAllSanPham();
            SanPham selectedProduct = null;
            
            for (SanPham sp : allProducts) {
                if (sp.getIdSanPham() == id) {
                    selectedProduct = sp;
                    break;
                }
            }
            
            if (selectedProduct != null) {
                txtTen.setText(selectedProduct.getTenSanPham());
                txtDonGia.setText(selectedProduct.getDonGia().toString());
                txtSoLuong.setText(String.valueOf(selectedProduct.getSoLuong()));
                txtMoTa.setText(selectedProduct.getMoTa() != null ? selectedProduct.getMoTa() : "");
                txtHinhAnh.setText(selectedProduct.getHinhAnh() != null ? selectedProduct.getHinhAnh() : "");
                txtThoiGianChuanBi.setText(String.valueOf(selectedProduct.getThoiGianChuanBi()));
                chkCoBanMangVe.setSelected(selectedProduct.isCoBanMangVe());
                chkCoBanTaiQuan.setSelected(selectedProduct.isCoBanTaiQuan());
                cboTrangThai.setSelectedItem(selectedProduct.getTrangThai());
                
                // Set danh mục
                for (int i = 0; i < cboDanhMuc.getItemCount(); i++) {
                    String item = cboDanhMuc.getItemAt(i);
                    if (item.startsWith(selectedProduct.getIdDanhMuc() + " - ")) {
